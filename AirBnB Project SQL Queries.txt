SELECT DISTINCT
host_name,
l.host_id,
l.id AS listing_id,
l.number_of_reviews,
l.name
FROM listings l
ORDER BY l.number_of_reviews DESC
LIMIT 10;

SELECT DISTINCT
l2.neighbourhood,
AVG(price) AS average_price
FROM listings_2 l2  
GROUP BY l2.neighbourhood  
ORDER BY average_price DESC
LIMIT 5;

SELECT
c.listing_id,
l.host_name,
l.host_id,
COUNT(c.listing_id) AS March_availability_count
FROM calendar c JOIN listings l ON c.listing_id = l.id
WHERE c.available = 't' AND c.date BETWEEN '2025-03-01' AND '2025-03-30'
GROUP BY c.listing_id
ORDER BY March_availability_count ASC 
LIMIT 10;

SELECT DISTINCT
l.host_name,
l.host_id,
l.calculated_host_listings_count,
l.reviews_per_month 
FROM listings l
ORDER BY l.calculated_host_listings_count ASC 
LIMIT 15;


SELECT
    l.host_id,
    l.host_name,
    AVG(l.review_scores_cleanliness) AS average_cleanliness_score
FROM listings l
GROUP BY l.host_id, l.host_name
HAVING AVG(l.review_scores_cleanliness) > 0
ORDER BY average_cleanliness_score ASC
LIMIT 15;
